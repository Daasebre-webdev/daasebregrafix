CREATE TABLE `users` (
  `id` INT(11) NOT NULL AUTO_INCREMENT,
  `google_id` VARCHAR(50) UNIQUE DEFAULT NULL,
  `name` VARCHAR(100) NOT NULL,
  `email` VARCHAR(100) NOT NULL UNIQUE,
  `password` VARCHAR(255) DEFAULT NULL,
  `picture` VARCHAR(255) DEFAULT NULL,
  `created_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` TIMESTAMP NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
  `is_verified` TINYINT(1) DEFAULT 0,
  `verify_token` VARCHAR(6) DEFAULT NULL,
  `verify_token_expires_at` DATETIME DEFAULT NULL,
  `last_login_at` TIMESTAMP NULL DEFAULT NULL,
  `login_method` ENUM('email','google') DEFAULT 'email',
  `status` ENUM('active','suspended','deleted') DEFAULT 'active',
  `agreed_to_terms` TINYINT(1) DEFAULT 0,
  PRIMARY KEY (`id`),
  UNIQUE KEY `unique_email` (`email`),
  UNIQUE KEY `unique_google_id` (`google_id`),
  INDEX `idx_verify_token` (`verify_token`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- For generated projects (already exists in your setup)
CREATE TABLE IF NOT EXISTS `projects` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `user_id` VARCHAR(255) NOT NULL,
  `title` VARCHAR(255) NOT NULL,
  `category` VARCHAR(100) DEFAULT 'Uncategorized',
  `description` TEXT,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- For bookmarks
CREATE TABLE IF NOT EXISTS `bookmarks` (
  `user_id` VARCHAR(255) NOT NULL,
  `project_id` INT NOT NULL,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`user_id`, `project_id`),
  FOREIGN KEY (`project_id`) REFERENCES `projects`(`id`) ON DELETE CASCADE
);
